#include <stdio.h>

int main() {
    char str[100], rev[100];
    int len = 0, i, j = 0;

    printf("Enter a word: ");
    scanf("%s", str);

    while (str[len] != '\0') {
        len++;
    }

    for (i = len - 1; i >= 0; i--) {
        rev[j] = str[i];
        j++;
    }
    rev[j] = '\0';

    printf("Reversed string: %s\n", rev);
    return 0;
}
